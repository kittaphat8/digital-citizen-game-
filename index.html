uiText = this.add.text(20, 20, '', { fontSize: '18px', fill: '#fff' }); bossBtn = this.add.rectangle(700, 40, 140, 40, 0xe74c3c).setInteractive(); this.add.text(700, 40, 'ไปสู้บอส ➔', { fontSize: '16px', fill: '#fff' }).setOrigin(0.5); cursors = this.input.keyboard.createCursorKeys(); this.input.on('pointerdown', (pointer) => { if (currentMode === 'FARM' && !isTalking && hero.body.touching.down && pointer.y > 100) { hero.setVelocityY(-550); } }); updateUI(); }
